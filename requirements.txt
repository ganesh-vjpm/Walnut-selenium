# ----------------------------------------------------------------------------
# 10. HOW TO ADD A NEW DEPENDENCY
# ----------------------------------------------------------------------------
#
# Step 1: Find the Maven coordinates at https://mvnrepository.com/
#         Example: groupId, artifactId, version
#
# Step 2: Open pom.xml and add inside <dependencies>...</dependencies>:
#
#         <dependency>
#             <groupId>com.example</groupId>
#             <artifactId>example-library</artifactId>
#             <version>1.0.0</version>
#             <scope>test</scope>    <!-- use "test" for test-only code -->
#         </dependency>
#
# Step 3: Optionally define a version property in <properties>:
#
#         <example.version>1.0.0</example.version>
#
#         and use it as:
#
#         <version>${example.version}</version>
#
# Step 4: Save pom.xml. Maven will download the dependency automatically.
#
# Step 5: Verify with:
#
#         mvn clean compile test-compile

# ----------------------------------------------------------------------------
# 11. HOW TO UPDATE A VERSION
# ----------------------------------------------------------------------------
#
# Change the version number in <properties> in pom.xml, for example:
#
#     <selenium.version>4.48.0</selenium.version>
#
# Then run:
#
#     mvn clean compile test-compile

# ----------------------------------------------------------------------------
# 12. HOW TO CHECK DOWNLOADED DEPENDENCIES
# ----------------------------------------------------------------------------
#
# Command:
#     mvn dependency:tree
#
# This prints all resolved dependencies and their transitive dependencies.

# ----------------------------------------------------------------------------
# 13. SCOPE MEANINGS
# ----------------------------------------------------------------------------
#
# compile   -> Available everywhere (main + test). Default scope.
# test      -> Only available in src/test/java and test execution.
# provided  -> Available at compile time, but provided at runtime (e.g., Lombok).
# runtime   -> Not needed at compile time, but needed at runtime.

# ----------------------------------------------------------------------------
# 14. CONFIGURATION AND ENVIRONMENT VARIABLES
# ----------------------------------------------------------------------------
#
# ConfigManager (src/main/java/com/walnut/automation/config/ConfigManager.java)
# resolves values in this priority order:
#
#   1. Java system property   -> -Dbase.url=https://your-app.com
#   2. OS environment variable -> BASE_URL=https://your-app.com
#   3. Properties file         -> config/qa.properties
#
# The active environment is selected by:
#   - system property: -Denvironment=uat
#   - env variable:    ENVIRONMENT=uat
#   - default:         qa
#
# Environment variable naming:
#   - For keys without dots: use the same uppercase name, e.g., BROWSER=firefox
#   - For keys with dots: use uppercase with underscores, e.g., BASE_URL=...
#
# Example GitHub Actions usage:
#   env:
#     ENVIRONMENT: qa
#     BROWSER: chrome
#     HEADLESS: true
#     MAX_RETRIES: 1
